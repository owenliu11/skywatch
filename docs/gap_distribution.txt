Reproduce:
WITH d AS (
 SELECT region, extract(epoch FROM time-lag(time) OVER
 (PARTITION BY icao24,region ORDER BY time)) s
 FROM state_vectors WHERE region IN ('bay_area','socal')
 AND time>now()-interval '6 hours'
)
SELECT region,count(*),percentile_disc(0.5) WITHIN GROUP (ORDER BY s) p50,
percentile_disc(0.9) WITHIN GROUP (ORDER BY s) p90,
percentile_disc(0.99) WITHIN GROUP (ORDER BY s) p99,max(s)
FROM d WHERE s>0 GROUP BY region;
